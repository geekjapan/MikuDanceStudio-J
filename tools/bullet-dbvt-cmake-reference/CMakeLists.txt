cmake_minimum_required(VERSION 3.21)
project(BulletDbvtCmakeReference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The reference requires native Windows x64 MSVC")
endif()
find_package(BULLET275 REQUIRED)
add_executable(dbvt_cmake_reference probe.cpp)
target_link_libraries(dbvt_cmake_reference PRIVATE BULLET275::BULLET275)
target_link_options(dbvt_cmake_reference PRIVATE /VERBOSE:LIB)
