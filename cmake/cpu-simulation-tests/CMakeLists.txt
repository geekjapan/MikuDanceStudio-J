cmake_minimum_required(VERSION 3.16)
project(MikuDanceStudioCpuSimulationTests LANGUAGES CXX)
include(CTest)
if(MSVC)
    add_compile_options(/Zc:alignedNew- /fp:precise)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-ffp-contract=off)
endif()
get_filename_component(MDS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)

# Build the exact vendored Bullet revision without Win32, a package server or
# configuring its demo/application projects. Keep the recipe's source lists.
set(BULLET_SOURCE_DIR "${MDS_SOURCE}/recipes/bullet275/bullet-src")
set(BULLET_PHYSICS_SOURCE_DIR "${BULLET_SOURCE_DIR}")
set(BULLET_VERSION 2.75)
set(BUILD_SHARED_LIBS OFF)
include_directories("${BULLET_SOURCE_DIR}/src")
foreach(component IN ITEMS LinearMath BulletCollision BulletDynamics)
    add_subdirectory("${BULLET_SOURCE_DIR}/src/${component}" "bullet/${component}")
endforeach()
add_library(mds_cpu_simulation
    "${MDS_SOURCE}/src/physics/physics_world_state.cpp"
    "${MDS_SOURCE}/src/physics/cpu_scene_simulation.cpp")
target_compile_features(mds_cpu_simulation PUBLIC cxx_std_17)
target_include_directories(mds_cpu_simulation PUBLIC "${MDS_SOURCE}/include")
target_link_libraries(mds_cpu_simulation PUBLIC BulletDynamics BulletCollision LinearMath)
foreach(test IN ITEMS physics_world_state cpu_scene_simulation)
    add_executable(${test}_test "${MDS_SOURCE}/tests/${test}_test.cpp")
    target_link_libraries(${test}_test PRIVATE mds_cpu_simulation)
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
